package engine

import (
	"context"
	"os"
	"path/filepath"
	"testing"

	"session-memory/internal/agent"
)

func TestAuditRebuildReportsDamagedRevision(t *testing.T) {
	r := newRig(t)
	tid, _ := r.put(991, 1)
	b, rev, err := r.eng.Track(context.Background(), r.ad, agent.NativeRef{Home: r.home, NativeID: tid}, "")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(filepath.Join(r.root, "store/revisions", rev.RevisionID+".json"), []byte("{broken"), 0600); err != nil {
		t.Fatal(err)
	}
	if err := r.st.Rebuild(context.Background()); err != nil {
		return
	}
	revs, err := r.st.Revisions(b.SessionID)
	if err != nil {
		t.Fatal(err)
	}
	if len(revs) == 0 {
		t.Fatal("Rebuild returned success but silently omitted the damaged revision still referenced by its binding")
	}
}
