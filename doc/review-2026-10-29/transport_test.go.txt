package transport

import (
	"context"
	"errors"
	"strings"
	"testing"
)

func TestAuditMirrorRejectsEqualLengthRemoteCollision(t *testing.T) {
	targets = map[string]string{}
	f := newFake(t)
	ctx := context.Background()
	local := NewFolder(t.TempDir())
	local.Settle = 0
	w := NewMirror(local, f.client(), "SessionMemory")
	if err := w.Put(ctx, "objects/ab/test.obj", strings.NewReader("aaaa")); err != nil {
		t.Fatal(err)
	}
	other := NewFolder(t.TempDir())
	other.Settle = 0
	m := NewMirror(other, f.client(), "SessionMemory")
	if err := m.Put(ctx, "objects/ab/test.obj", strings.NewReader("bbbb")); !errors.Is(err, ErrExists) {
		t.Fatalf("remote same-name different bytes were accepted because their lengths match: %v", err)
	}
}

func TestAuditMultiMirrorRetriesMissingKey(t *testing.T) {
	mm, _, f2, _ := twoAccounts(t)
	ctx := context.Background()
	f2.mu.Lock()
	f2.quota = true
	f2.mu.Unlock()
	if err := mm.PutKey(ctx, []byte("test-key")); err != nil {
		t.Fatal(err)
	}
	f2.mu.Lock()
	f2.quota = false
	f2.mu.Unlock()
	if err := mm.CatchUp(ctx); err != nil {
		t.Fatal(err)
	}
	_, found, err := mm.Accounts[1].M.GetKey(ctx)
	if err != nil {
		t.Fatal(err)
	}
	if !found {
		t.Fatal("partial key publication reported success and was never retried on recovered backup")
	}
}
