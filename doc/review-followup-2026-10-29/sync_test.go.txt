// The only format difference between a pre-fix written journal and this
// fixture is the newly introduced Planned field. Keep Written and backups.
func followupStripPlanned(t *testing.T, b *node) {
	t.Helper()
	paths, err := filepath.Glob(filepath.Join(b.root, "journal", "*.json"))
	if err != nil || len(paths) != 1 {
		t.Fatalf("journals: %v %v", paths, err)
	}
	data, err := os.ReadFile(paths[0])
	if err != nil {
		t.Fatal(err)
	}
	var record map[string]json.RawMessage
	if err := json.Unmarshal(data, &record); err != nil {
		t.Fatal(err)
	}
	var payload map[string]json.RawMessage
	if err := json.Unmarshal(record["data"], &payload); err != nil {
		t.Fatal(err)
	}
	if _, ok := payload["planned"]; !ok {
		t.Fatal("setup: new journal lacks planned")
	}
	delete(payload, "planned")
	record["data"], err = json.Marshal(payload)
	if err != nil {
		t.Fatal(err)
	}
	data, err = json.Marshal(record)
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(paths[0], data, 0600); err != nil {
		t.Fatal(err)
	}
}

// Positive check for the second A03 guard: a rollout appearing while staging
// must be refused even if the earlier Plan saw a clean target.
func TestFollowupApplyRefusesExtraRolloutAfterPlanning(t *testing.T) {
	_, b, sid, target := updatedPair(t, true)
	ctx := context.Background()
	graph, err := b.st.Graph(sid)
	if err != nil || len(graph.Heads()) != 1 {
		t.Fatalf("graph: %v", err)
	}
	rev, err := b.st.LoadRevision(graph.Heads()[0])
	if err != nil {
		t.Fatal(err)
	}
	source, err := b.st.SnapshotReader(ctx, rev.NativeSnapshotRef)
	if err != nil {
		t.Fatal(err)
	}
	before, err := os.ReadFile(target)
	if err != nil {
		t.Fatal(err)
	}
	stem := filepath.Base(target)
	extra := filepath.Join(filepath.Dir(target), stem[:len(stem)-len(".jsonl")]+"_11111111-1111-4111-8111-111111111111.jsonl")
	const marker = "{\"type\":\"event_msg\",\"payload\":{\"message\":\"LATE_LOCAL_ONLY\"}}\n"
	hook := &reviewHookSnapshot{SnapshotReader: source, hook: func() {
		if err := os.WriteFile(extra, []byte(marker), 0600); err != nil {
			t.Fatal(err)
		}
	}}
	in := agent.WriteInput{SessionID: sid, RevisionID: rev.RevisionID, Mode: agent.ModeNative, Source: hook, SourceProfile: rev.AgentProfile, Target: rev.AgentProfile, TargetState: agent.TargetState{Home: b.home}}
	w := &codex.Writer{ClientClosed: true}
	plan, err := w.Plan(ctx, in)
	if err != nil {
		t.Fatal(err)
	}
	tx, err := b.st.Begin(ctx, store.KindInstall, sid, "")
	if err != nil {
		t.Fatal(err)
	}
	if _, err := w.Apply(ctx, in, plan, tx); errcode.CodeOf(err) != errcode.BaselineChanged {
		t.Fatalf("late rollout was not refused: %v", err)
	}
	if got, err := os.ReadFile(target); err != nil || !bytes.Equal(got, before) {
		t.Fatalf("main changed: %v", err)
	}
	if got, err := os.ReadFile(extra); err != nil || string(got) != marker {
		t.Fatalf("extra changed: %v", err)
	}
	if err := tx.Finish(); err != nil {
		t.Fatal(err)
	}
}

func TestFollowupLegacyRecoveryPreservesPostCrashProgress(t *testing.T) {
	for _, existing := range []bool{false, true} {
		label := "create"
		if existing {
			label = "append"
		}
		t.Run(label, func(t *testing.T) {
			_, b, sid, target := updatedPair(t, existing)
			crashInstall(t, b, sid)
			followupStripPlanned(t, b)
			f, err := os.OpenFile(target, os.O_APPEND|os.O_WRONLY, 0600)
			if err != nil {
				t.Fatal(err)
			}
			if _, err = f.WriteString("{\"type\":\"event_msg\",\"payload\":{\"message\":\"LEGACY_AFTER_CRASH\"}}\n"); err != nil {
				t.Fatal(err)
			}
			if err := f.Close(); err != nil {
				t.Fatal(err)
			}
			want, err := os.ReadFile(target)
			if err != nil {
				t.Fatal(err)
			}
			reopened := reopenWithGuard(t, b)
			_, recoverErr := reopened.Recover(context.Background())
			got, readErr := os.ReadFile(target)
			if readErr != nil || !bytes.Equal(got, want) {
				t.Fatalf("upgraded recovery erased continuation from legacy journal: recover=%v read=%v", recoverErr, readErr)
			}
			if errcode.CodeOf(recoverErr) != errcode.RecoveryNeeded {
				t.Fatalf("changed legacy target should retain journal and require attention: %v", recoverErr)
			}
		})
	}
}

func TestFollowupLegacyUntouchedJournalStillRollsBack(t *testing.T) {
	_, b, sid, target := updatedPair(t, true)
	before, err := os.ReadFile(target)
	if err != nil {
		t.Fatal(err)
	}
	crashInstall(t, b, sid)
	followupStripPlanned(t, b)
	reopened := reopenWithGuard(t, b)
	if _, err := reopened.Recover(context.Background()); err != nil {
		t.Fatal(err)
	}
	got, err := os.ReadFile(target)
	if err != nil || !bytes.Equal(got, before) {
		t.Fatalf("legacy control rollback: %v", err)
	}
}
